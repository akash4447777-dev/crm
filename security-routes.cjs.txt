'use strict';
const C=require('./security-core.cjs');
module.exports=function routes({service:S,dbGet,users,saveUsers,out,body}){
 return async function handle(req,res,u,M,me){
 const send=(code,data)=>{out(res,code,data);return true;},superOnly=()=>{if(!C.isSuper(me))C.fail(403,'Super Admin required.');};
 if(!u.startsWith('/data')&&!u.startsWith('/security'))return false;
 if(!C.isSuper(me)&&!C.scope(me).size)C.fail(403,'No company assigned. Ask Super Admin to assign company access.');
 if(M==='GET'&&u==='/data/ver'){const {row}=await S.readData(me);return send(200,{ver:row?.ver||0,securityVersion:15});}
 if(M==='GET'&&u==='/data'){const {row,env}=await S.readData(me);return send(200,{ver:row?.ver||0,data:row?C.project(env.data,me):null,securityVersion:15,user:C.pub(me)});}
 if(M==='PUT'&&u==='/data'){C.writer(me);const input=await body(req),{row,env}=await S.readData(me);if((row?.ver||0)!==+input.base)return send(409,{conflict:true,ver:row?.ver||0,data:row?C.project(env.data,me):null});const prepared=C.prepare(env.data,input.data,me,env.security),next={__crmEnvelope:1,data:prepared.data,security:prepared.security};const result=await S.commit('data',row?.ver||0,next,me,'dataset-update',prepared.changes.concat(C.securityDifferences(env.security,next.security)));if(!result.ok){const latest=await S.readData(me);return send(409,{conflict:true,ver:latest.row?.ver||0,data:C.project(latest.env.data,me)});}return send(200,{ok:true,ver:result.ver,data:C.project(next.data,me),pendingApprovals:prepared.pending,securityVersion:15});}
 if(M==='GET'&&u==='/security'){const {row,env}=await S.readData(me),seal=await S.latest('data');return send(200,{version:15,user:C.pub(me),initialized:!!seal,ver:row?.ver||0,companies:(env.data.cos||[]).map((c,i)=>({id:i+1,name:c.m||c.n})).filter(c=>C.can(me,c.id)),thresholds:Object.fromEntries(Object.entries(env.security.thresholds||{}).filter(([c])=>C.can(me,c))),approvals:(env.security.approvals||[]).filter(a=>C.can(me,a.company)).slice(-500).reverse(),limitsUnit:'paise',defaultApprovalThreshold:0});}
 if(M==='POST'&&u==='/security/initialize'){superOnly();const old=await dbGet('data','ver,value');if(old){const env=C.unpack(old.value);await S.mutate(me,'security-baseline',()=>env);}const ur=await dbGet('users','ver,value');if(ur){await S.checkStored('users',ur,true);const result=await S.commit('users',ur.ver,ur.value,me,'users-baseline',[]);if(!result.ok)C.fail(409,'Users changed during initialization. Try again.');}return send(200,{ok:true,message:'Current state signed as baseline. Earlier history cannot be reconstructed.'});}
 if(M==='GET'&&u==='/security/users'){superOnly();return send(200,{users:(await users()).list.map(C.pub)});}
 if(M==='POST'&&u==='/security/access'){superOnly();const input=await body(req),{env}=await S.readData(me);if(!['admin','viewer'].includes(input.role))C.fail(422,'Role must be Admin or Viewer.');if(!Array.isArray(input.companies)||input.companies.some(c=>!Number.isInteger(c)||c<=0||!env.data.cos?.[c-1]))C.fail(422,'Select valid company IDs.');const result=await saveUsers(list=>{const user=list.find(x=>x.id===input.id);if(!user)return {err:'User not found.'};if(user.role==='super')return {err:'Super Admin access cannot be downgraded here.'};user.role=input.role;user.companies=[...new Set(input.companies)];user.av=(user.av||0)+1;return {ok:true};},me,'company-access-update');return send(result.err?422:200,result);}
 if(M==='POST'&&u==='/security/settings'){superOnly();const input=await body(req);await S.mutate(me,'approval-limit-update',env=>{const id=String(input.company);if(!Number.isInteger(+id)||!env.data.cos?.[+id-1]||!C.money(input.threshold))C.fail(422,'Valid company and nonnegative rupee limit required.');env.security.thresholds[id]=Math.round(+input.threshold*100);return env;});return send(200,{ok:true});}
 if(M==='GET'&&u==='/security/audit'){const query=new URL(req.url,'http://crm.local').searchParams;return send(200,await S.audit(me,query.get('before')));}
 if(M==='POST'&&u==='/security/verify'){superOnly();return send(200,await S.verify());}
 if(M==='POST'&&u==='/security/restore'){superOnly();const input=await body(req);const restored=await S.restore(me,input.seq,input.index);return send(200,{ok:true,ver:restored.ver});}
 if(M==='POST'&&u==='/security/decision'){superOnly();const input=await body(req);if(!['approve','reject'].includes(input.decision))C.fail(422,'Choose approve or reject.');const result=await S.mutate(me,'advance-'+input.decision,env=>C.decision(env,input.id,input.decision==='approve',me));return send(200,{ok:true,ver:result.ver});}
 return send(404,{error:'Unknown security endpoint.'});
 };
};
