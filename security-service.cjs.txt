'use strict';
const C=require('./security-core.cjs');
module.exports=function create({url,headers,dbGet,key,fetcher=fetch}){
 const ready=()=>{if(!key||key.length<32)C.fail(503,'Set a separate AUDIT_SIGNING_KEY (at least 32 random characters) and run migrations/015_security.sql.');};
 async function query(path){ready();const r=await fetcher(url+'/rest/v1/'+path,{headers:headers()});if(!r.ok)C.fail(503,'Security database unavailable. Apply migrations/015_security.sql and check the server service-role key. No write was accepted.');return r.json();}
 async function latest(stream){return (await query('crm_audit?stream=eq.'+stream+'&order=seq.desc&limit=1'))[0]||null;}
 function verifyEvent(row){if(!row||!C.safeEqual(C.mac(row.event_text,key),row.event_mac))C.fail(503,'Audit signature mismatch. Writes blocked; investigate the ledger.');const event=JSON.parse(row.event_text);if(event.previousHead!==row.prev_hash||row.hash!==C.hash(row.prev_hash+'\n'+row.event_text+'\n'+row.event_mac))C.fail(503,'Signed audit chain link mismatch.');return event;}
 async function checkStored(stream,row,allowBaseline=false){const last=await latest(stream);if(!last){if(row&&!allowBaseline)C.fail(503,'Security baseline missing. Super Admin must initialize Security Center before access.');return false;}const event=verifyEvent(last);if(event.stateHash!==C.hash(row?.value??null))C.fail(503,'Unauthorized database change detected: state does not match its signed audit seal. Writes blocked.');return true;}
 async function readData(user){const row=await dbGet('data','ver,value');await checkStored('data',row,C.isSuper(user));return {row,env:C.unpack(row?.value)};}
 async function commit(stream,base,value,user,action,changes=[],extra={}){ready();const stateText=C.canonical(value);for(let attempt=0;attempt<6;attempt++){const previousHead=(await query('crm_audit_head?select=hash'))[0]?.hash||'',event={id:require('node:crypto').randomUUID(),stream,at:new Date().toISOString(),actor:user?{id:user.id,name:user.name,role:user.role}:{id:'registration',name:'Account registration',role:'public'},action,changes,stateHash:C.hash(stateText),...extra,previousHead};const eventText=C.canonical(event);const r=await fetcher(url+'/rest/v1/rpc/crm_secure_commit',{method:'POST',headers:headers(),body:JSON.stringify({p_key:stream,p_base:base,p_state_text:stateText,p_event_text:eventText,p_mac:C.mac(eventText,key)})});if(!r.ok)C.fail(503,'Atomic audited save failed. Check migration, database and signing-key configuration. Nothing was committed.');const result=await r.json();if(result.headConflict)continue;return result;}C.fail(409,'Audit ledger is busy. Retry the save.');}
 async function mutate(user,action,fn){for(let i=0;i<6;i++){const {row,env}=await readData(user),next=await fn(C.copy(env));const result=await commit('data',row?.ver||0,next,user,action,C.differences(env.data,next.data).concat(C.securityDifferences(env.security,next.security)),{securityChanges:!C.equal(env.security,next.security)});if(result.ok)return {env:next,ver:result.ver};}C.fail(409,'Concurrent changes could not be merged. Refresh and try again.');}
 async function audit(user,before){const cursor=before?String(before):null;if(cursor&&!/^\d+$/.test(cursor))C.fail(422,'Invalid audit cursor.');const rows=await query('crm_audit?order=seq.desc&limit=100'+(cursor?'&seq=lt.'+cursor:''));const events=[];for(const row of rows){const event=verifyEvent(row);if(!C.isSuper(user)){event.changes=(event.changes||[]).filter(ch=>C.can(user,ch.company));if(!event.changes.length)continue;delete event.securityChanges;}events.push({seq:row.seq,recordedAt:row.recorded_at,event,hash:row.hash});}return {events,nextCursor:rows.length===100?String(rows[rows.length-1].seq):null};}
 async function verify(){let cursor=0,previous='',count=0,last;for(;;){const rows=await query('crm_audit?order=seq.asc&limit=1000&seq=gt.'+cursor);for(const row of rows){verifyEvent(row);const expected=C.hash(previous+'\n'+row.event_text+'\n'+row.event_mac);if(row.prev_hash!==previous||row.hash!==expected)C.fail(503,'Audit hash-chain verification failed at event '+row.seq);previous=row.hash;cursor=row.seq;last=row;count++;}if(rows.length<1000)break;}const head=(await query('crm_audit_head?select=hash'))[0];if((head?.hash||'')!==previous)C.fail(503,'Audit head mismatch; history may have been truncated.');for(const stream of ['data','users']){const row=await dbGet(stream,'ver,value');if(row)await checkStored(stream,row,false);}return {ok:true,events:count,head:previous,checkedAt:new Date().toISOString(),lastEvent:last?.recorded_at||null};}
 async function restore(user,seq,index){if(!C.isSuper(user))C.fail(403,'Only Super Admin can restore deletions.');if(!/^\d+$/.test(String(seq))||!Number.isInteger(index)||index<0)C.fail(422,'Invalid event/change reference.');const row=(await query('crm_audit?seq=eq.'+seq+'&limit=1'))[0],event=verifyEvent(row),change=event.changes?.[index];if(!change||change.after!==null||change.before===null||![...C.arrays,...C.maps].includes(change.collection))C.fail(422,'This event is not a restorable deletion.');return mutate(user,'restore-deletion',env=>{const data=env.data,col=change.collection;if(C.arrays.includes(col)){if((data[col]||[]).some(r=>String(r.id)===String(change.id)))C.fail(409,'Record ID is already in use. Nothing overwritten.');data[col]||=[];data[col].push(C.copy(change.before));}else{data[col]||={};if(Object.hasOwn(data[col],change.id))C.fail(409,'Entry already exists. Nothing overwritten.');data[col][change.id]=C.copy(change.before);}C.validate({},data);return env;});}
 return {ready,query,latest,verifyEvent,checkStored,readData,commit,mutate,audit,verify,restore};
};
