'use strict';
const C=require('./security-core.cjs'),R=require('./statements-core.js'),{randomUUID}=require('node:crypto');
function exportOptions(raw){const o=R.options(raw);for(const k of ['excelFreeze','excelFilter','excelWrap','zipManifest'])if(o[k]!==undefined&&typeof o[k]!=='boolean')C.fail(422,'Invalid export checkbox: '+k);if(o.zipCompression!==undefined&&!['fast','balanced','small'].includes(o.zipCompression))C.fail(422,'Invalid ZIP compression.');return o;}
function centerValue(input){if(!input||typeof input!=='object'||Array.isArray(input))C.fail(422,'Invalid export center settings.');const p=input.preferences??{};if(!p||typeof p!=='object'||Array.isArray(p))C.fail(422,'Invalid export center preference object.');if(p.remember!==undefined&&typeof p.remember!=='boolean'||p.onboarded!==undefined&&typeof p.onboarded!=='boolean'||p.mode!==undefined&&!['quick','advanced'].includes(p.mode))C.fail(422,'Invalid export center preference.');const presets=input.presets??[],history=input.history??[];if(!Array.isArray(presets)||presets.length>20||!Array.isArray(history)||history.length>25)C.fail(422,'Keep at most 20 custom presets and 25 history entries.');const ids=new Set();const cleanPresets=presets.map(v=>{if(!v||typeof v.id!=='string'||!/^[a-zA-Z0-9_-]{1,80}$/.test(v.id)||ids.has(v.id)||typeof v.name!=='string'||!v.name.trim()||v.name.length>80)C.fail(422,'Invalid preset ID or name.');ids.add(v.id);return {id:v.id,name:v.name.trim(),options:exportOptions(v.options),updatedAt:typeof v.updatedAt==='string'?v.updatedAt.slice(0,40):''};});const cleanHistory=history.map(v=>{if(!v||typeof v.id!=='string'||!/^[a-zA-Z0-9_-]{1,80}$/.test(v.id)||typeof v.name!=='string'||v.name.length>240||!['pdf','xlsx','zip','csv','all'].includes(v.format)||!['generated','failed','cancelled'].includes(v.status)||typeof v.createdAt!=='string'||!Number.isFinite(Date.parse(v.createdAt))||!Number.isSafeInteger(v.size)||v.size<0)C.fail(422,'Invalid download history entry.');return {id:v.id,name:v.name,format:v.format,status:v.status,createdAt:v.createdAt,size:v.size,message:typeof v.message==='string'?v.message.slice(0,500):''};});const value={preferences:{remember:p.remember!==false,onboarded:p.onboarded===true,mode:p.mode==='advanced'?'advanced':'quick'},presets:cleanPresets,history:cleanHistory};if(Buffer.byteLength(JSON.stringify(value),'utf8')>8000000)C.fail(422,'Preset metadata is larger than 8 MB. Reduce large preset logos or remove unused presets.');return value;}
module.exports=function({service:S,out,body,saveUsers}){return async(req,res,u,M,me)=>{
 if(!u.startsWith('/statements'))return false;
 if(!['/statements/report','/statements/preferences','/statements/center'].includes(u)||u==='/statements/report'&&M!=='POST'||['/statements/preferences','/statements/center'].includes(u)&&!['GET','PUT'].includes(M)){out(res,404,{error:'Unknown statements endpoint.'});return true;}
 if(!C.isSuper(me)&&!C.scope(me).size)C.fail(403,'No company assigned.');
 if(u==='/statements/center'&&M==='GET'){out(res,200,{center:me.statementExportCenter||null,storage:'signed shared database'});return true;}
 if(u==='/statements/preferences'&&M==='GET'){const p=me.statementExportPreferences||null;out(res,200,{preferences:p,storage:'signed shared database'});return true;}
 const input=await body(req);
 if(u==='/statements/center'){if(!saveUsers)C.fail(503,'Export center storage is not configured.');if(input.userId&&input.userId!==me.id)C.fail(403,'Only your own export center may be changed.');if(!Number.isSafeInteger(input.base)||input.base<0)C.fail(422,'Invalid center version.');const value=centerValue(input.center);let saved;await saveUsers(list=>{const user=list.find(v=>v.id===me.id);if(!user||user.status!=='active'||(user.av||0)!==(me.av||0))C.fail(401,'Your account changed. Sign in again.');if((user.statementExportCenter?.version||0)!==input.base)C.fail(409,'Export center changed on another device. Reload the center before saving; nothing was overwritten.');saved={schema:19,version:input.base+1,updatedAt:new Date().toISOString(),...value};user.statementExportCenter=saved;return {ok:true};},me,'export-center-save');out(res,200,{center:saved,storage:'signed shared database'});return true;}

 if(u==='/statements/preferences'){
  if(!saveUsers)C.fail(503,'Statement preference storage is not configured on this backend.');if(input.userId&&input.userId!==me.id)C.fail(403,'Only your own export preferences may be changed.');
  const options=exportOptions(input.options);if(!Number.isSafeInteger(input.base)||input.base<0)C.fail(422,'Invalid preference version. Refresh settings before saving.');
  const {env}=await S.readData(me);const ids=new Map((env.data.w||[]).map(w=>[String(w.id),w]));for(const id of options.companies)if(!env.data.cos?.[+id-1]||!C.can(me,id))C.fail(403,'Saved company selection is not authorized.');for(const id of options.workerIds)if(!ids.has(id)||!C.can(me,ids.get(id).c))C.fail(403,'Saved worker selection is missing or not authorized.');
  let saved;await saveUsers(list=>{const current=list.find(x=>x.id===me.id);if(!current||current.status!=='active'||(current.av||0)!==(me.av||0))C.fail(401,'Your account changed. Sign in again.');const prior=current.statementExportPreferences;if((prior?.version||0)!==input.base)C.fail(409,'Your defaults changed on another device. Reload settings before saving; nothing was overwritten.');saved={schema:17,version:input.base+1,updatedAt:new Date().toISOString(),options};current.statementExportPreferences=saved;return {ok:true};},me,'statement-preferences-save');out(res,200,{preferences:saved,storage:'signed shared database'});return true;
 }
if(input.data)C.fail(422,'Report data must come from the canonical server, not the browser.');
 const raw=input.options||me.statementExportPreferences?.options;if(!raw)C.fail(422,'Save your Advance Statement defaults first.');const o=exportOptions(raw),format=input.format||'pdf';if(!['pdf','preview','print','zip','csv','xlsx','all'].includes(format))C.fail(422,'Unsupported statement format.');
 for(let attempt=0;attempt<6;attempt++){
  const {row,env}=await S.readData(me);if(!row)C.fail(422,'No shared CRM records. Import an authorized backup first.');
  for(const id of o.companies)if(!env.data.cos?.[+id-1]||!C.can(me,id))C.fail(403,'Requested company is not authorized.');
  const byId=new Map((env.data.w||[]).map(w=>[String(w.id),w]));for(const id of o.workerIds){const w=byId.get(id);if(!w||!C.can(me,w.c))C.fail(403,'Requested worker is not authorized.');}
  const exportId=randomUUID(),sourceVersion=row.ver,report=R.build(C.project(env.data,me),o,{exportId,sourceVersion,source:'canonical shared database (authorized snapshot)'});
  const changes=report.companies.map(g=>({collection:'statement-export',id:exportId,company:g.id,before:null,after:{action:'authorization only, not confirmed file delivery',format,from:o.from,to:o.to,workers:g.workers,closingPaise:g.closing,template:o.template,paper:o.paper,language:o.language,transactionType:o.transactionType,fields:o.fields,dateFormat:o.dateFormat,sourceVersion}}));
  const result=await S.commit('data',row.ver,env,me,'statement-export-authorized',changes,{exportId,sourceVersion});
  if(result.ok){out(res,200,{report,auditVersion:result.ver});return true;}
 }
 C.fail(409,'The shared data changed repeatedly. Refresh and retry the report.');
};};
